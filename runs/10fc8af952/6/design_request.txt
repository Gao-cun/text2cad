工程基线说明
- 原始需求：我要一个桌面手机支架，适合竖放看视频，结构尽 量简洁，方便 3D 打印。
- 设计摘要：采用一体化L型或斜面支撑结构，底部宽大稳固，背部设有限位与防滑特征，整体避免复杂装配与悬垂结构，以优化FDM打印成功率。
- 对象类型：手机支架
- 目标尺寸（mm）：长 120.00，宽 100.00，高 130.00
- 材料：PLA，杨氏模量 3500.00 MPa，泊松比 0.36
- 固定边界盒：[0.0, 0.0, 0.0, 120.0, 100.0, 10.0]
- 加载边界盒：[30.0, 30.0, 50.0, 90.0, 70.0, 110.0]
- 试探载荷向量（N）：[0.0, 0.0, -4.9]
- 设计目标：稳固支撑竖屏手机; 结构简洁易打印; 减少或消除打印支撑; 防滑防倾倒
- 风格关键词：极简, 几何, 实用主义, 一体化
- 视觉要求：表面平整光滑，减少层纹视觉干扰; 边缘采用适度倒角处理以提升手感与安全性; 无复杂浮雕或外露纹理，保持极简几何外观; 整体呈现单色/哑光质感
- 默认假设：默认适配主流智能手机（宽度70-80mm，厚度8-10mm）; 假设打印方向为Z轴向上，底座平放于构建板，无需额外支撑结构; 试探性载荷按单部手机约0.5kg重力垂直向下施加于背部支撑区域; 结构为单件一体成型，不包含活动铰链、螺丝或二次装配
- 执行说明：以上边界盒、试探载荷和材料参数是当前求解后端的默认分析上下文，不是造型本身的唯一解；若你为了更符合真实物体语义而调整几何，请在 self_check_notes 中说明。
- 约束：必须生成可执行的 CadQuery 代码，几何应可网格化、可渲染、可进入后续力学和视觉审查。

实现边界与优先级

- 不要退化成无语义方块，除非用户需求本身就是简单实体。

```python
import cadquery as cq
import math

def build_model():
    L, W, H = 120.0, 100.0, 130.0
    base_h = 12.0
    back_thick = 8.0
    rib_thick = 6.0

    # 1. 底座 (增加0.05mm重叠确保布尔运算成功)
    base = cq.Workplane("XY").box(L, W, base_h + 0.05)

    # 2. 背板 (XZ平面绘制，Y向拉伸)
    bp_x0, bp_z0 = 25.0, base_h
    bp_x1, bp_z1 = 50.0, H
    back_prof = (
        cq.Workplane("XZ")
        .moveTo(bp_x0, bp_z0 - 0.05)
        .lineTo(bp_x1, bp_z1)
        .lineTo(bp_x1 - back_thick, bp_z1)
        .lineTo(bp_x0 - back_thick, bp_z0 - 0.05)
        .close()
    )
    back_plate = back_prof.workplane(offset=-W/2).extrude(W)

    # 3. 加强筋 (三角肋)
    rib_x0, rib_x1 = 10.0, bp_x0 - back_thick
    rib_z_top = 60.0
    rib_prof = (
        cq.Workplane("XZ")
        .moveTo(rib_x0, base_h - 0.05)
        .lineTo(rib_x1, base_h - 0.05)
        .lineTo(rib_x1, rib_z_top)
        .close()
    )
    rib = rib_prof.workplane(offset=-rib_thick/2).extrude(rib_thick)

    # 4. 顶部限位挡边
    stop_x = bp_x1 - back_thick
    stop = cq.Workplane("XY").box(3.0, 80.0, 8.0).translate((stop_x, 0, bp_z1 - 0.05 + 4.0))

    # 5. 前端侧挡边
    guide1 = cq.Workplane("XY").box(4.0, 10.0, 6.0).translate((110.0, 37.5, base_h - 0.05 + 3.0))
    guide2 = cq.Workplane("XY").box(4.0, 10.0, 6.0).translate((110.0, -37.5, base_h - 0.05 + 3.0))

    # 合并主体
    model = base.union(back_plate).union(rib).union(stop).union(guide1).union(guide2)

    # 6. 背部防滑特征 (交叉网格凹槽)
    face_cx = (bp_x0 - back_thick + bp_x1 - back_thick) / 2
    face_cz = (bp_z0 + bp_z1) / 2
    angle = math.degrees(math.atan2(bp_z1 - bp_z0, bp_x1 - bp_x0))
    anti_slip = (
        cq.Workplane("XZ")
        .transformed(offset=(face_cx, 0, face_cz), rotate=(0, angle, 0))
        .rarray(3.0, 3.0, 12, 16)
        .rect(1.5, 1.5)
        .extrude(0.3)
    )
    model = model.cut(anti_slip)

    # 7. 全局倒角处理
    model = model.edges(">Z").fillet(1.5)
    model = model.edges("<Z").fillet(1.5)
    model = model.edges("|Y").fillet(2.0)
    model = model.edges("<X").fillet(2.0)
    model = model.edges(">X").fillet(1.5)
    model = model.clean()

    # 8. 平移至第一象限
    bbox = model.val().BoundingBox()
    model = model.translate((-bbox.xmin, -bbox.ymin, -bbox.zmin))
    model = model.clean()
    return model
```
