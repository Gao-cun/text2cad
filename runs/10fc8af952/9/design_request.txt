工程基线说明
- 原始需求：我要一个桌面手机支架，适合竖放看视频，结构尽 量简洁，方便 3D 打印。
- 设计摘要：采用一体化L型或斜面支撑结构，底部宽大稳固，背部设有限位与防滑特征，整体避免复杂装配与悬垂结构，以优化FDM打印成功率。
- 对象类型：手机支架
- 目标尺寸（mm）：长 120.00，宽 100.00，高 130.00
- 材料：PLA，杨氏模量 3500.00 MPa，泊松比 0.36
- 固定边界盒：[0.0, 0.0, 0.0, 120.0, 100.0, 10.0]
- 加载边界盒：[30.0, 30.0, 50.0, 90.0, 70.0, 110.0]
- 试探载荷向量（N）：[0.0, 0.0, -4.9]
- 设计目标：稳固支撑竖屏手机; 结构简洁易打印; 减少或消除打印支撑; 防滑防倾倒
- 风格关键词：极简, 几何, 实用主义, 一体化
- 视觉要求：表面平整光滑，减少层纹视觉干扰; 边缘采用适度倒角处理以提升手感与安全性; 无复杂浮雕或外露纹理，保持极简几何外观; 整体呈现单色/哑光质感
- 默认假设：默认适配主流智能手机（宽度70-80mm，厚度8-10mm）; 假设打印方向为Z轴向上，底座平放于构建板，无需额外支撑结构; 试探性载荷按单部手机约0.5kg重力垂直向下施加于背部支撑区域; 结构为单件一体成型，不包含活动铰链、螺丝或二次装配
- 执行说明：以上边界盒、试探载荷和材料参数是当前求解后端的默认分析上下文，不是造型本身的唯一解；若你为了更符合真实物体语义而调整几何，请在 self_check_notes 中说明。
- 约束：必须生成可执行的 CadQuery 代码，几何应可网格化、可渲染、可进入后续力学和视觉审查。

```python
import cadquery as cq
import math

def build_model():
    L, W, H = 120.0, 100.0, 130.0
    base_h = 12.0
    back_thick = 8.0
    rib_thick = 5.0

    # 1. 主体轮廓 (XZ平面)
    main_prof = (
        cq.Workplane("XZ")
        .moveTo(0, 0)
        .lineTo(L, 0)
        .lineTo(L, base_h)
        .lineTo(45, H - 5)
        .lineTo(45 - back_thick, H - 5)
        .lineTo(25, base_h)
        .lineTo(0, base_h)
        .close()
    )
    # 修复：移除原代码中引发语法/解析错误的 .workplane(offset=...) 调用，直接对闭合轮廓拉伸
    main_body = main_prof.extrude(W)

    # 2. 安全倒角：避开底面边缘 (<Z)，防止破坏打印接触面
    main_body = main_body.edges().not("<Z").fillet(2.0).clean()

    # 3. 根部加强筋 (三角肋)
    rib_x0, rib_x1 = 10.0, 25.0
    rib_z_top = 60.0
    rib_prof = (
        cq.Workplane("XZ")
        .moveTo(rib_x0, base_h)
        .lineTo(rib_x1, base_h)
        .lineTo(rib_x1, rib_z_top)
        .close()
    )
    rib = rib_prof.extrude(rib_thick)
    rib = rib.edges().not("<Z").fillet(1.5).clean()
    model = main_body.union(rib, glue=True).clean()

    # 4. 前端侧挡边
    guide_w, guide_h, guide_d = 4.0, 6.0, 10.0
    guide_y = 37.5
    guide1 = cq.Workplane("XY").box(guide_w, guide_d, guide_h).translate((110.0, guide_y, base_h + guide_h/2))
    guide2 = cq.Workplane("XY").box(guide_w, guide_d, guide_h).translate((110.0, -guide_y, base_h + guide_h/2))
    guides = guide1.union(guide2).edges().not("<Z").fillet(1.0).clean()
    model = model.union(guides, glue=True).clean()

    # 5. 顶部限位挡边
    stop_x = 45 - back_thick
    stop = cq.Workplane("XY").box(3.0, 80.0, 8.0).translate((stop_x - 1.5, 0, H - 5 + 4.0))
    stop = stop.edges().not("<Z").fillet(1.0).clean()
    model = model.union(stop, glue=True).clean()

    # 6. 背部防滑纹理 (横向凹槽阵列)
    face_cx = (25 + 45 - back_thick) / 2
    face_cz = (base_h + H - 5) / 2
    angle = math.degrees(math.atan2((H - 5) - base_h, 45 - 25))
    anti_slip = (
        cq.Workplane("XZ")
        .transformed(offset=(face_cx, 0, face_cz), rotate=(0, angle, 0))
        .rarray(5.0, 5.0, 5, 6)
        .rect(2.5, 0.8)
        .extrude(0.6)
    )
    model = model.cut(anti_slip).clean()

    # 7. 平移至第一象限原点
    bbox = model.BoundingBox()
    model = model.translate((-bbox.xmin, -bbox.ymin, -bbox.zmin)).clean()
    return model
```
